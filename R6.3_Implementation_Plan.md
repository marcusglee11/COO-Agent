## Executive Summary

R6.3 Fix Packet introduces **breaking changes** to AMU₀ structure, canonical hashing, and platform requirements. This implementation plan addresses all clarified requirements and provides a structured approach to the transition.

### Key Breaking Changes from R6.2:
1. **AMU₀ ID Derivation**: Pure hash-based (no `amu0_id.txt` file)
2. **Canonical Hash**: Excludes rollback log (resolves circular dependency)
3. **Platform**: Linux-only enforcement at import level
4. **PATH Pinning**: Fixed `/usr/bin:/bin` (no dynamic PATH)
5. **Initialization**: Explicit `initialize_runtime()` required before all operations

#### A2. Canonical Hash Update ✅ CLARIFIED
**Requirement**: Deterministic traversal, mtime normalization, explicit exclusions

**Implementation**:
```python
def calculate_canonical_hash(amu0_dir: str) -> bytes:
    """
    Calculate canonical SHA-256 hash of AMU0 bundle.
    Excludes: rollback_log.jsonl, rollback_log.sig, signature.sig, *.tmp
    """
    hasher = hashlib.sha256()
    
    # Sorted traversal
    for root, dirs, files in sorted(os.walk(amu0_dir)):
        dirs[:] = sorted(dirs)
        for filename in sorted(files):
            # Skip excluded files
            if filename in ['rollback_log.jsonl', 'rollback_log.sig', 
                          'signature.sig'] or filename.endswith('.tmp'):
                continue
            
            filepath = os.path.join(root, filename)
            rel_path = os.path.relpath(filepath, amu0_dir)
            
            # Hash: path + mtime (normalized) + content
            hasher.update(rel_path.encode('utf-8'))
            hasher.update(struct.pack('<Q', 315532800))  # Normalized mtime
            
            with open(filepath, 'rb') as f:
                hasher.update(f.read())
    
    return hasher.digest()
```

#### A4. Unified AMU₀ Verification ✅ CLARIFIED
**Requirement**: Single `verify_amu0_complete()` function with typed result

**Implementation**:
```python
from dataclasses import dataclass

@dataclass
class VerificationResult:
    canonical_hash: bytes
    amu0_id: str
    has_rollback_log: bool

def verify_amu0_complete(amu0_path: str) -> VerificationResult:
    """
    Verify AMU0 integrity. Raises GovernanceError on any failure.
    
    Checks:
    - Directory structure
    - Required files (fs_snapshot, snapshot_manifest.json, etc.)
    - Canonical hash computation
    - Bundle signature validation
    - Rollback log integrity (if present)
    - Derived ID matches tracker
    
    Returns: VerificationResult (only on full success)
    Raises: GovernanceError (on any failure)
    """
    # 1. Check structure
    if not os.path.isdir(amu0_path):
        raise GovernanceError(f"AMU0 path is not a directory: {amu0_path}")
    
    # 2. Check required files
    required = ['fs_snapshot', 'snapshot_manifest.json', 'pinned_context.json']
    for req in required:
        if not os.path.exists(os.path.join(amu0_path, req)):
            raise GovernanceError(f"Missing required file/dir: {req}")
    
    # 3. Compute canonical hash
    canonical_hash = calculate_canonical_hash(amu0_path)
    amu0_id = derive_amu0_id(canonical_hash)
    
    # 4. Verify bundle signature
    sig_path = os.path.join(amu0_path, 'signature.sig')
    if not os.path.exists(sig_path):
        raise GovernanceError("Missing bundle signature")
    
    with open(sig_path, 'rb') as f:
        signature = f.read()
    
    if not Signature.verify_data(canonical_hash, signature, 
                                 get_ceo_public_key_path()):
        raise GovernanceError("AMU0 bundle signature verification failed")
    
    # 5. Verify rollback log (if present)
    has_rollback_log = False
    log_path = os.path.join(amu0_path, 'rollback_log.jsonl')
    if os.path.exists(log_path):
        # Verify log signature and chain
        verify_rollback_log_integrity(amu0_path)
        has_rollback_log = True
    
    # 6. Success
    return VerificationResult(
        canonical_hash=canonical_hash,
        amu0_id=amu0_id,
        has_rollback_log=has_rollback_log
    )
```

```python
import platform

def capture_hardware_context() -> Dict[str, str]:
    """
    Capture hardware context. Raises GovernanceError if unavailable.
    """
    # Kernel version
    kernel_version = platform.release()
    if not kernel_version:
        raise GovernanceError("Cannot capture kernel version")
    
    # CPU microcode from /proc/cpuinfo
    microcode_values = []
    try:
        with open('/proc/cpuinfo', 'r') as f:
            for line in f:
                if line.startswith('microcode'):
                    value = line.split(':')[1].strip()
                    if value:
                        microcode_values.append(value)
    except Exception as e:
        raise GovernanceError(f"Cannot read /proc/cpuinfo: {e}")
    
    if not microcode_values:
        raise GovernanceError("No CPU microcode values found in /proc/cpuinfo")
    
    # Deterministic representation
    cpu_microcode = "|".join(sorted(set(microcode_values)))
    
    return {
        "kernel_version": kernel_version,
        "cpu_microcode": cpu_microcode
    }
```

```python
# coo_runtime/runtime/init.py (new file)

_initialized_amu0_path: Optional[str] = None

def initialize_runtime(amu0_path: str) -> None:
    """
    Initialize runtime with pinned context from AMU0.
    Must be called once per process before any deterministic operations.
    
    Rules:
    - First call: Initialize with given amu0_path
    - Second call with SAME amu0_path: No-op
    - Call with DIFFERENT amu0_path: Raise GovernanceError
    
    Raises: GovernanceError on initialization failure or multiple AMU0s
    """
    global _initialized_amu0_path
    
    if _initialized_amu0_path is not None:
        if _initialized_amu0_path == amu0_path:
            return  # Idempotent for same AMU0
        else:
            raise GovernanceError(
                f"Runtime already initialized with {_initialized_amu0_path}. "
                f"Cannot re-initialize with {amu0_path}."
            )
    
    # 1. Load pinned context
    context_path = os.path.join(amu0_path, 'pinned_context.json')
    with open(context_path, 'r') as f:
        context = json.load(f)
    
    # 2. Verify hardware
    current_hw = capture_hardware_context()
    if current_hw['kernel_version'] != context.get('kernel_version'):
        raise GovernanceError(
            f"Kernel version mismatch: "
            f"Current={current_hw['kernel_version']}, "
            f"Pinned={context.get('kernel_version')}"
        )
    if current_hw['cpu_microcode'] != context.get('cpu_microcode'):
        raise GovernanceError(
            f"CPU microcode mismatch: "
            f"Current={current_hw['cpu_microcode']}, "
            f"Pinned={context.get('cpu_microcode')}"
        )
    
    # 3. Seed RNG
    import random
    seed = context.get('rng_seed', 'DETERMINISTIC_SEED_DEFAULT')
    random.seed(seed)
    
    # 4. Enable libfaketime (if configured in context)
    # This would set LD_PRELOAD, but actual time pinning is subprocess-level
    
    # 5. Verify time pinning
    expected_time = context.get('mock_time')
    if expected_time:
        _verify_time_pinning(context, expected_time)
    
    # Mark as initialized
    _initialized_amu0_path = amu0_path

def assert_initialized() -> None:
    """Assert that runtime has been initialized. For internal checks."""
    if _initialized_amu0_path is None:
        raise GovernanceError(
            "Runtime not initialized. Call initialize_runtime(amu0_path) first."
        )
```

```python
TIME_PIN_TOLERANCE_SECONDS = 1.1

def _verify_time_pinning(env: Dict, expected_time: str) -> None:
    """
    Verify time pinning by spawning subprocess.
    Raises GovernanceError if difference > 1.1 seconds.
    """
    # Convert expected_time to timestamp
    from datetime import datetime
    expected_ts = datetime.fromisoformat(expected_time).timestamp()
    
    # Spawn subprocess with pinned environment
    import subprocess
    result = subprocess.run(
        [sys.executable, '-c', 'import time; print(time.time())'],
        env=env,
        capture_output=True,
        text=True,
        check=True
    )
    
    actual_ts = float(result.stdout.strip())
    diff = abs(actual_ts - expected_ts)
    
    if diff > TIME_PIN_TOLERANCE_SECONDS:
        raise GovernanceError(
            f"Time pinning verification failed. "
            f"Expected: {expected_time} ({expected_ts}), "
            f"Actual: {actual_ts}, "
            f"Difference: {diff}s > {TIME_PIN_TOLERANCE_SECONDS}s"
        )
```

```python
# coo_runtime/util/subprocess.py (NEW)

PINNED_PATH = "/usr/bin:/bin"

def run_pinned_subprocess(
    cmd: List[str],
    amu0_path: str,
    **kwargs
) -> subprocess.CompletedProcess:
    """
    Run subprocess with pinned context from AMU0.
    
    All subprocess calls MUST use this helper.
    Direct usage of subprocess.run/check_output/Popen is FORBIDDEN.
    
    Args:
        cmd: Command and arguments
        amu0_path: Path to AMU0 for pinned context
        **kwargs: Additional subprocess.run arguments
    
    Returns: CompletedProcess
    Raises: GovernanceError if tool not found or execution fails
    """
    # Load pinned context
    context_path = os.path.join(amu0_path, 'pinned_context.json')
    with open(context_path, 'r') as f:
        context = json.load(f)
    
    # Build pinned environment
    pinned_env = {}
    
    # 1. Fixed PATH
    pinned_env['PATH'] = PINNED_PATH
    
    # 2. Allowed env vars from context
    for key, value in context.get('env_vars', {}).items():
        pinned_env[key] = value
    
    # 3. LD_PRELOAD for libfaketime (if configured)
    if context.get('libfaketime_enabled'):
        pinned_env['LD_PRELOAD'] = '/usr/lib/faketime/libfaketime.so.1'
        pinned_env['FAKETIME'] = context.get('mock_time', '2025-11-28 00:00:00')
    
    # Override any env in kwargs
    if 'env' in kwargs:
        kwargs.pop('env')
    
    try:
        return subprocess.run(cmd, env=pinned_env, **kwargs)
    except FileNotFoundError as e:
        raise GovernanceError(
            f"Required tool '{cmd[0]}' not found in pinned PATH {PINNED_PATH}. "
            f"Ensure tool is installed in /usr/bin or /bin."
        ) from e
```

```python
# coo_runtime/util/crypto.py

class Signature:
    """Unified signature protocol for all COO Runtime signing operations."""
    
    @staticmethod
    def sign_data(data: bytes, private_key_path: str) -> bytes:
        """Sign arbitrary bytes. Returns raw signature bytes."""
        from nacl.signing import SigningKey
        with open(private_key_path, 'rb') as f:
            sk = SigningKey(f.read())
        return sk.sign(data).signature
    
    @staticmethod
    def verify_data(data: bytes, signature: bytes, public_key_path: str) -> bool:
        """Verify signature over bytes. Returns True on success, False on failure."""
        from nacl.signing import VerifyKey
        from nacl.exceptions import BadSignatureError
        try:
            with open(public_key_path, 'rb') as f:
                vk = VerifyKey(f.read())
            vk.verify(data, signature)
            return True
        except BadSignatureError:
            return False
    
    @staticmethod
    def sign_file(filepath: str, private_key_path: str) -> bytes:
        """
        Read file contents and sign them.
        Returns signature bytes.
        Caller writes to <filepath>.sig if desired.
        """
        with open(filepath, 'rb') as f:
            data = f.read()
        return Signature.sign_data(data, private_key_path)
    
    @staticmethod
    def verify_file(filepath: str, signature: bytes, public_key_path: str) -> bool:
        """Read file contents and verify signature bytes."""
        with open(filepath, 'rb') as f:
            data = f.read()
        return Signature.verify_data(data, signature, public_key_path)
```
